module splash;

import milktea;
import xray;
import glaze;
import std::io;
import std::core::string;

extern fn double c3_sin(double x);
extern fn double c3_cos(double x);


struct SplashModel {
	sz frame;
	sz width;
	sz height;
	// Cell grid for direct rendering
	xray::Cell[] view_cells;
	sz view_cells_len;
}

fn milktea::Cmd SplashModel.init(SplashModel* self) @dynamic {
	return milktea::tick();
}


fn milktea::Cmd SplashModel.update(SplashModel* self, milktea::Msg msg) @dynamic {
	if (msg.kind == milktea::MsgKind.WINDOW_SIZE) {
		self.width = msg.window_size.width;
		self.height = msg.window_size.height;
		// Reallocate cell grid
		mem::free(self.view_cells.ptr);
		sz rows = self.height;
		self.view_cells_len = (sz)self.width * (sz)rows;
		self.view_cells = mem::new_array(xray::Cell, self.view_cells_len);
		return null;
	}
	if (msg.kind == milktea::MsgKind.TICK) {
		self.frame++;
		return milktea::tick();
	}
	if (msg.kind != milktea::MsgKind.KEY) return null;
	return milktea::quit();
}

fn sz clamp_color(sz v) {
	if (v < 0) return 0;
	if (v > 255) return 255;
	return v;
}

struct RgbColor {
	sz r;
	sz g;
	sz b;
}

fn RgbColor hsv_to_rgb(double h, double s, double v) {
	double hh = h * 6.0;
	if (hh < 0) hh = 0;
	sz i     = (sz)hh;
	double ff = hh - (double)i;
	double p  = v * (1.0 - s);
	double q  = v * (1.0 - s * ff);
	double t  = v * (1.0 - s * (1.0 - ff));

	double r = 0;
	double g = 0;
	double b = 0;

	switch (i % 6) {
		case 0: r = v; g = t; b = p;
		case 1: r = q; g = v; b = p;
		case 2: r = p; g = v; b = t;
		case 3: r = p; g = q; b = v;
		case 4: r = t; g = p; b = v;
		case 5: r = v; g = p; b = q;
		default: r = v; g = p; b = p;
	}

	return { .r = clamp_color((sz)(r * 255)), .g = clamp_color((sz)(g * 255)), .b = clamp_color((sz)(b * 255)) };
}

}

}

fn milktea::Node* SplashModel.view(SplashModel* self) @dynamic {
	sz w = self.width;
	sz h = self.height;
	if (w < 2) w = 80;
	if (h < 2) h = 24;

	double t     = (double)self.frame * 0.03;
	double angle = 0.7854 + t * 0.1;

	sz rows = h;

	// Reallocate if size changed
	sz needed = (sz)w * (sz)rows;
	if (needed != self.view_cells_len) {
		mem::free(self.view_cells.ptr);
		self.view_cells_len = needed;
		self.view_cells = mem::new_array(xray::Cell, needed);
	}

	sz pixel_rows = h * 2;
	for (sz y = 0; y < rows; y++) {
		sz fy_top     = y * 2;
		sz fy_bot     = y * 2 + 1;
		sz row_offset = y * w;

		for (sz x = 0; x < w; x++) {
			double fx = (double)x / (double)w;

			double ca = c3_cos(angle);
			double sa = c3_sin(angle);

			// Top pixel color
			double yt     = (double)fy_top / (double)pixel_rows;
			double proj_t = (fx - 0.5) * ca + (yt - 0.5) * sa + 0.5;
			double vt     = 0.6 + 0.4 * c3_sin(fx * 3.0 + t * 0.7 + yt * 2.0);
			double ht     = (proj_t * 0.4 + t * 0.1) % 1.0;
			if (ht < 0) ht += 1.0;
			RgbColor top = hsv_to_rgb(ht, 0.8, vt);

			// Bottom pixel color
			double yb     = (double)fy_bot / (double)pixel_rows;
			double proj_b = (fx - 0.5) * ca + (yb - 0.5) * sa + 0.5;
			double vb     = 0.6 + 0.4 * c3_sin(fx * 3.0 + t * 0.7 + yb * 2.0);
			double hb     = (proj_b * 0.4 + t * 0.1) % 1.0;
			if (hb < 0) hb += 1.0;
			RgbColor bot = hsv_to_rgb(hb, 0.8, vb);

			self.view_cells[row_offset + x] = xray::half_block(
				xray::color_rgb((char)top.r, (char)top.g, (char)top.b),
				xray::color_rgb((char)bot.r, (char)bot.g, (char)bot.b));
		}
	}

	return milktea::grid(self.view_cells, w, rows);
}

fn int main() {
	SplashModel m = { };
	return milktea::@run(&m, { .alt_screen = true });
}
